; Function to test custom placeholde paths or variables defined on the .INI file
TestPlaceHolderPaths() {
	BaseString := IniRead(IniSettingsFilePath, "UserDefinedPaths", "TestMe", "")
	BaseString := ReplacePlaceholderStrings(BaseString)
	MsgBox(BaseString)
}


; Function to bring mouse cursor to the center of current active application monitor
; Useful in multi-monitor scenarios where you switch between windows and lose track of the cursor
MouseComeHome() {
	CoordMode("Mouse", "Window")
	WinGetPos(, , &ActWinWidth, &ActWinHeight, "A")
	MouseMove(ActWinWidth / 2, ActWinHeight / 2, 0)
}


; App-specific hot Keys definition for KiTTY
; ===================================================

	#HotIf WinActive("ahk_class PuTTYConfigBox")
		; Validation to enable  start button quick key press on Putty, very convenient 
		; Putty Window info
		; KiTTY Configuration
		; ahk_class PuTTYConfigBox
		; ahk_exe KITTY_~1.EXE
		
		; Control for start button
		; ClassNN:	Button2
		; Text:	Start
		!s:: {
			ControlFocus("Button2")
			ControlSend("{Space}", "Button2")
		}

		!a:: {
			ControlFocus("Edit1")
		}
	#HotIf

	#HotIf WinActive("ahk_class KiTTY")
		^!Up::Send("{PgUp}")
		^!Down::Send("{PgDn}")

		; Duplicate session: Opens KiTTY system menu (Alt+Space), then presses 'd'
		!d:: {
		    KeyWait("Alt")
		    Send("!{Space}d")
		}

		; Copy all text to clipboard: Opens KiTTY system menu (Alt+Space), then presses 'o'
		^!c:: {
		    KeyWait("Ctrl")
		    KeyWait("Alt")
		    Send("!{Space}o")
		}

		; Start WinSCP
		^!s:: {
		    Send("+{F3}")
		}

		; Clear window text and scrollback: Ctrl+L, then Alt+Space followed by 'l'
		^!l:: {
		    Send("^l")
		    KeyWait("Ctrl")
		    KeyWait("Alt")
		    Sleep(50)
		    Send("!{Space}l")
		}
	#HotIf

	; Automatically focus ACCEPT button for KiTTY new server connection
	#HotIf WinActive("ahk_class #32770")
		!a:: {
			ControlFocus("Button2")
			ControlSend("{Space}", "Button2")
		}
	#HotIf
